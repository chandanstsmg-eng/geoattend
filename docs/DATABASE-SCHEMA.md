## Views

### attendance_work

```sql
SELECT ar.id AS attendance_id,
    ar.employee_id,
    ar.shift_id,
    ar.shift_date,
    ar.workplace_id,
    ar.session_id,
    s.id IS NOT NULL AND s.ended_at IS NULL AS in_progress,
    sched.minutes AS scheduled_minutes,
    away.minutes AS away_minutes,
        CASE
            WHEN s.id IS NULL THEN ar.manual_worked_minutes
            ELSE GREATEST(0, round(EXTRACT(epoch FROM reg.ends_at - reg.starts_at) / 60::numeric)::integer - away.minutes - brk.unpaid_minutes)
        END AS worked_minutes,
        CASE
            WHEN s.ended_at IS NOT NULL THEN GREATEST(0, round(EXTRACT(epoch FROM s.scheduled_end_at - s.ended_at) / 60::numeric)::integer)
            ELSE NULL::integer
        END AS ended_early_minutes,
    brk.minutes AS break_minutes,
    brk.unpaid_minutes AS unpaid_break_minutes,
    ot.id AS overtime_id,
        CASE
            WHEN ot.id IS NULL THEN NULL::text
            WHEN ot.ended_at IS NULL THEN 'RUNNING'::text
            ELSE ot.status
        END AS overtime_status,
    COALESCE(ot.claimed, 0) AS overtime_claimed_minutes,
    COALESCE(
        CASE
            WHEN ot.status = 'APPROVED'::text THEN ot.approved_minutes
            ELSE NULL::integer
        END, 0) AS overtime_minutes
   FROM attendance_records ar
     JOIN shifts sh ON sh.id = ar.shift_id
     LEFT JOIN tracking_sessions s ON s.id = ar.session_id
     CROSS JOIN LATERAL ( SELECT GREATEST(ar.recorded_at, s.scheduled_start_at) AS starts_at,
            LEAST(COALESCE(s.ended_at, now()), s.scheduled_end_at) AS ends_at) reg
     CROSS JOIN LATERAL ( SELECT
                CASE
                    WHEN s.id IS NOT NULL THEN round(EXTRACT(epoch FROM s.scheduled_end_at - s.scheduled_start_at) / 60::numeric)::integer
                    ELSE round(EXTRACT(epoch FROM
                    CASE
                        WHEN sh.end_time > sh.start_time THEN sh.end_time - sh.start_time
                        ELSE sh.end_time - sh.start_time + '24:00:00'::interval
                    END) / 60::numeric)::integer
                END AS minutes) sched
     CROSS JOIN LATERAL ( SELECT COALESCE(round(sum(GREATEST(0::numeric, EXTRACT(epoch FROM LEAST(COALESCE(a.ended_at, s.ended_at, now()), reg.ends_at) - GREATEST(a.started_at, reg.starts_at)))) / 60::numeric), 0::numeric)::integer AS minutes
           FROM absence_intervals a
          WHERE a.session_id = ar.session_id) away
     CROSS JOIN LATERAL ( SELECT COALESCE(round(sum(GREATEST(0::numeric, EXTRACT(epoch FROM LEAST(COALESCE(b.ended_at, s.ended_at, now()), reg.ends_at) - b.started_at))) / 60::numeric), 0::numeric)::integer AS minutes,
            COALESCE(round(sum(GREATEST(0::numeric, EXTRACT(epoch FROM LEAST(COALESCE(b.ended_at, s.ended_at, now()), reg.ends_at) - b.started_at))) FILTER (WHERE NOT b.paid) / 60::numeric), 0::numeric)::integer AS unpaid_minutes
           FROM session_breaks b
          WHERE b.session_id = ar.session_id) brk
     LEFT JOIN LATERAL ( SELECT o.id,
            o.status,
            o.ended_at,
            o.approved_minutes,
            round(EXTRACT(epoch FROM COALESCE(o.ended_at, now()) - o.started_at) / 60::numeric)::integer AS claimed
           FROM overtime_entries o
          WHERE o.session_id = ar.session_id) ot ON true
  WHERE ar.verification_status <> 'VOIDED'::text;
```
